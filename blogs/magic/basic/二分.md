## 1.数据范围

[原题链接](https://www.acwing.com/problem/content/791/)

>给定一个按照升序排列的长度为 n的整数数组，以及 q个查询。

>对于每个查询，返回一个元素 k的起始位置和终止位置（位置从 0开始计数）。

>如果数组中不存在该元素，则返回 -1 -1。 

### 老模版

```c
#include<iostream>
using namespace std;
const int N = 100005;

int n,q;
int a[N];


int main()
{
    cin>>n>>q;
    for(int i = 0 ; i < n ; i++)
        cin>>a[i];
    for(int i = 0 ; i < q ; i++)
        {   
            int key;
            cin>>key;
            int r =n-1 ,l = 0;
            while(l < r)
                {
                   int m = (r + l)/2;
                    if(a[m] >= key) r = m;      //先找左端点，对于中间范围来讲，端点为key,而m是这段范围的中间
                    else                        //所以肯定找的事比左端点大的即 a[m] >= key
                        l = m + 1;              //至于为什么是r = m，还是第一句话,key < a[m]，要找的在中间的左边，所以动右指针
                }                               //else 后面的要么 l = m+1,要么r = m - 1,上面的要么 r =m 要么 l= m
                if(a[r] == key)                 //最后弹出的结果就是r
                {     
                 cout<<r<<" ";
                 r = n - 1;
                 while(l < r)
                    {
                    int m = (r + l+1)/2;
                    if(a[m] <= key) l = m;//同理的，要是l = m ，则上面的m必须加一，即,m=(l+r +1)/2
                    else
                        r = m - 1;
                     }
                    cout<<l<<endl;      //最后弹出的结果是l
                }
                else cout<<"-1 -1"<<endl;    
        }
    return 0;
}
```

### 新模板
[模板来源](https://blog.csdn.net/WJPnb1/article/details/126360962?spm=1001.2014.3001.5502)

新的模板不用考虑mid +- 1的情况了。可以试试这个
```c
#include<iostream>
using namespace std;

const int N = 1e5 + 5;
int a[N];

int main()
{
    int n , m;
    cin >> n >> m;
    for(int i = 0 ; i < n ; i ++)
        cin >> a[i];
    
    for(int i = 0 ; i < m ; i ++)
    {
        int x ;
        cin >> x ;
        int l = -1 , r = n ,mid;                //l ，r 不用考虑mid +/- 1的版本，l,r的初始值分别为-1和n
        while(l + 1 != r)                       //这个退出条件是l和r相邻时，比l < r的条件少一步。
        {
            mid = l + r >> 1;
            if(a[mid] < x)l = mid;              
            else r = mid;
        }
        if(a[r] != x)cout<< "-1 ";
        else cout << r << " ";                  //返回的结果取l还是r就看等号在哪里(暂定，还未总结通用的结论)
        
        l = -1 , r  = n ;
        while(l + 1 != r)
        {
            mid = l + r >> 1;
            if(a[mid] <= x) l = mid;
            else r = mid;
        }
        if(a[l] != x)cout<< "-1 "<<endl;
        else cout << l << " "<<endl;
    }
    return 0;
}
```

## 2.数的三次方根
[数的三次方根](https://www.acwing.com/problem/content/792/)

之前都是对有序的数组进行查找，这次变成了计算数值，二分并非只有一个查找。

>给定一个浮点数 n，求它的三次方根。

```c
#include<iostream>
using namespace std;

int main()
{
    double n ;
    cin >> n;
    
    double l = -100.0,r = 100.0;    //数据范围在+-10000之内，那么立方根自然在-100~100
    double mid;
    while(r - l > 1e-8)             //浮点数是连续的，让l、r不断地逼近，达到精度要求即可
    {
        mid = (l + r) /2.0;             
        if(mid * mid * mid < n) l = mid;    //二分查找同款思路，只是关键字判断需要改，条件取等号与否并不会影响结果
        else r = mid;                       //由于浮点数是连读的，不需要+-1
    }
    printf("%.6lf" , r);            //调整一下输出格式
    return 0;
}
```

## 3.x的平方根
[原题:力扣69.x的平方根](https://leetcode.cn/problems/sqrtx/description/?envType=problem-list-v2&envId=binary-search)

>给你一个非负整数 x ，计算并返回 x 的 算术平方根 。

>由于返回类型是整数，结果只保留 整数部分 ，小数部分将被 舍去 。

>注意：不允许使用任何内置指数函数和算符，例如 pow(x, 0.5) 或者 x ** 0.5 。

看似很简单，但是对于我这种初学者来说还是比较难搞的，虽然未来只要用sqrt()就能搞出来，但是对于二分的学习来说，还是得好好看一下的。

```c
int mySqrt(int x) {
        long long l = -1 , r = x + 1 , mid;     //这里也是新模板,x+1是能取到x
        while(r - l > 0)                        //因为题目要求整数，条件l < r
        {
            mid = (l + r) / 2;
            if(mid * mid ==x) return mid;       //因为mid*mid可能会很大，要longlong
            else if(mid * mid < x)l = mid;
            else r = mid;
        }
        return  r;
    }
```

## 4.搜索二维矩阵

[74.搜索二维矩阵](https://leetcode.cn/problems/search-a-2d-matrix/submissions/697275437/?envType=problem-list-v2&envId=binary-search)

>给你一个满足下述两条属性的 m x n 整数矩阵：

>1.每行中的整数从左到右按非严格递增顺序排列。

>2.每行的第一个整数大于前一行的最后一个整数。

>给你一个整数 target ，如果 target 在矩阵中，返回 true ；否则，返回 false 。

显然，我们可以纯暴力遍历这个二维矩阵，这里顺便优化一下采用二分查找

### 暴力
```c
bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int n = matrix[0].size();
        int m = matrix.size();
        
        while(m --)//从二维数组最下面的一行开始向上
        {
            int l = -1 , r = n , mid = 0;
            while(l + 1 != r)
            {
                mid = (l + r) / 2;
                if(matrix[m][mid] == target)    //中间对mid进行了判断
                {
                    return true;
                }
                if(matrix[m][mid] < target) l = mid;
                else r = mid;
            }
            //if(matrix[m][r] == target) return true;
            //该行代码存在越界风险，而且多余，r的值可能为n，但是n会越界
            //关于多余，因为二分时，有对mid所对应的值进行判断是否相等
        }
        return false;
    }
```
### 两次二分查找

因为各行之间也是有序的，可以按照各行的第一个进行二分查找，找到目标所在的行，再对这个行进行二分查找。

```c
bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int m = matrix.size();
        int n = matrix[0].size();
        int l = -1 , r = m , mid = 0;

        while(l + 1 != r)
        {
            mid = (l + r) / 2;
            if(matrix[mid][0] == target) return true;
            if(matrix[mid][0] < target) l = mid;    //l的最终结果就是最后一个值 <目标值
            else r = mid;
        }
        if(l < 0) return false;             //处理一下个别特殊数据
        int x = l;                          //为什么取l，这需要根据题目来分析。
                                            //
        int ll = -1 , rr = n;
        mid = 0;
        while(ll + 1 != rr)
        {
            mid = (ll + rr) / 2;
            if(matrix[x][mid] == target) return true;
            if(matrix[x][mid] < target ) ll = mid;
            else rr = mid;
        }

        return false;
    }

```
